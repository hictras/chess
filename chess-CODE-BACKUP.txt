# Chess Online — Render deployment

A responsive, real-time chess game for two devices. Node.js + Express + Socket.IO + chess.js.

## Features
- Create a private room or join with a room code
- Real-time moves synchronized between both players
- Server validates all moves with chess.js
- White/Black assignment, player names, move list, captured pieces
- Legal move highlighting, check/checkmate/stalemate/draw handling
- Flip board for Black, resign, offer draw, rematch request
- Reconnection to the same room using a per-tab session-stored player token (while the server process/room remains alive)
- Render health endpoint at `/health`

## Run locally
Requires Node.js 20+.

```bash
npm install
npm start
```

Open http://localhost:3000 in two browsers/devices. For two devices outside your local network, deploy it to Render.

## Deploy to Render
1. Create a GitHub repository and upload all files in this folder.
2. In Render, choose **New + → Blueprint** and select the repository (it reads `render.yaml`), or create a **Web Service** manually.
3. For manual setup:
   - Runtime: Node
   - Build command: `npm install`
   - Start command: `npm start`
4. Deploy and open the `https://your-service-name.onrender.com` address.
5. Share the deployed URL with the other player. Create a room, then share its room code.

## Important limitations
- This starter uses in-memory rooms. Rooms and games are lost if the service restarts or sleeps; a free service may have cold starts.
- It is intended for casual private games, not rated or money games.
- Browser refresh reconnection works while the room still exists and the player token is retained. For persistent rooms across restarts or multiple server instances, add Redis/database storage and a Socket.IO adapter.
- The website can be embedded in another site using an iframe pointing to the deployed URL, subject to the host site's iframe/CSP rules.

      if (typeof ms !== "number" || ms < 0) return "";

    function renderBoardFromFen(fen) {
      const lm = computeLastMove(currentStep);
      const g = new Chess(fen);

      }));

          boardEl.appendChild(cell);
        }

    function computeLastMove(step) {
      if (step === 0 || moveTimes.length === 0) return { from: null, to: null };
      const g = new Chess();
      let last = null;
      for (let i = 0; i < step; i++) {
        try { last = g.move(moves[i]); } catch(e) { break; }
      }
      return { from: last ? last.from : null, to: last ? last.to : null };
    }

    function computeFenAtStep(step) {
      const g = new Chess();
      for (let i = 0; i < step; i++) {
        try { g.move(moves[i]); } catch(e) { break; }
      }
      return g.fen();
    }

    function computeExactUsed() {
      let wUsed = 0;
      let bUsed = 0;
      for (let i = 0; i < currentStep; i++) {
        const prev = i > 0 ? moveTimes[i - 1] : 0;
        const now = moveTimes[i];
        if (typeof now !== "number") return { wUsed, bUsed };
        const spent = Math.max(0, (now - prev) / 1000);
        if (i % 2 === 0) wUsed += spent;
        else bUsed += spent;
      }
      return { wUsed, bUsed };
    }

    function paintClocks() {
      const whiteRemaining = Math.max(0, 600 - liveWUsed);
      const blackRemaining = Math.max(0, 600 - liveBUsed);
      $("whiteClock").textContent = fmtClock(Math.round(whiteRemaining));
      $("blackClock").textContent = fmtClock(Math.round(blackRemaining));
    }

    function updateClocks() {
      if (!gameMeta) return;
      const ex = computeExactUsed();
      liveWUsed = ex.wUsed;
      liveBUsed = ex.bUsed;
      paintClocks();
    }

    function startTicker() {
      stopTicker();
      tickTimer = setInterval(() => {
        // Who is on the clock? Next move's color.
        // currentStep = number of moves already played. Next move index = currentStep.
        // Even index = white's move (white on clock). Odd = black on clock.
        if (currentStep >= moves.length) return;
        if (currentStep % 2 === 0) liveWUsed += 1;
        else liveBUsed += 1;
        paintClocks();
      }, 1000);
    }

    function stopTicker() {
      if (tickTimer) { clearInterval(tickTimer); tickTimer = null; }
    }

    function render() {
      renderBoardFromFen(computeFenAtStep(currentStep));
      $("moveLabel").textContent = "Move " + currentStep + " / " + moves.length;
      if (currentStep > 0) {
        const idx = currentStep - 1;
        const t = moveTimes[idx] !== undefined ? " (" + fmtMoveTime(moveTimes[idx]) + ")" : "";
        $("currentMove").textContent = moves[idx] + t;
      } else {
        $("currentMove").textContent = "—";
      }
      $("prevBtn").disabled = currentStep === 0;
      $("nextBtn").disabled = currentStep === moves.length;
      updateClocks();
      const rows = $("moveList").querySelectorAll(".move-pair");
      const activeRow = Math.floor((currentStep - 1) / 2);
      rows.forEach((el, i) => {
        el.style.background = i === activeRow ? "rgba(155,201,108,.15)" : "";
      });
      if (playing) {
        updateClocks();
        startTicker();
      }
    }

    function renderMoveList() {
      const list = $("moveList");
      list.innerHTML = "";
      if (!moves.length) {
        list.innerHTML = '<div class="empty-moves">No moves recorded</div>';
        $("moveCount").textContent = "0 moves";
        return;
      }
      for (let i = 0; i < moves.length; i += 2) {
        const pair = document.createElement("div");
        pair.className = "move-pair";
        const t1 = moveTimes[i] !== undefined ? '<span class="move-time">(' + fmtMoveTime(moveTimes[i]) + ')</span>' : "";
        const t2 = moveTimes[i+1] !== undefined ? '<span class="move-time">(' + fmtMoveTime(moveTimes[i+1]) + ')</span>' : "";
        pair.innerHTML =
          '<span class="move-no">' + (Math.floor(i/2)+1) + '.</span>' +
          '<span class="move-san">' + moves[i] + t1 + '</span>' +
          '<span class="move-san">' + (moves[i+1] || '') + t2 + '</span>';
        list.appendChild(pair);
      }
      $("moveCount").textContent = moves.length + " moves";
    }

    function step(d) {
      currentStep = Math.max(0, Math.min(moves.length, currentStep + d));
      render();
    }

    function togglePlay() {
      if (playing) {
        clearTimeout(playTimer);
        playing = false;
        $("playBtn").textContent = "▶ Play";
        stopTicker();
        return;
      }
      if (currentStep >= moves.length) currentStep = 0;
      playing = true;
      $("playBtn").textContent = "⏸ Pause";
      updateClocks();
      startTicker();
      scheduleNext();
    }

    function scheduleNext() {
      if (!playing) return;
      if (currentStep >= moves.length) {
        playing = false;
        $("playBtn").textContent = "▶ Play";
        stopTicker();
        return;
      }
      let delay;
      if (speedMode === 'realtime') {
        const prevTime = currentStep > 0 ? (moveTimes[currentStep - 1] || 0) : 0;
        const thisTime = moveTimes[currentStep] || (prevTime + 800);
        delay = Math.max(150, thisTime - prevTime);
      } else {
        delay = 800;
      }
      playTimer = setTimeout(() => {
        currentStep++;
        render();
        scheduleNext();
      }, delay);
    }

        renderMoveList();
        render();
      } catch (e) {

      }

function cleanName(value) {
  return String(value || "Player")
    .trim()
    .replace(/[<>]/g, "")
    .slice(0, 24) || "Player";
}

function makeCode() {
  const chars = "ABCDEFGHJKLMNPQRSTUVWXYZ23456789";
  let code;

  do {
    code = Array.from(
      { length: 6 },
      () => chars[crypto.randomInt(chars.length)]
    ).join("");
  } while (rooms.has(code));

  return code;
}

function publicState(room) {
  const game = room.game;

  const history = game.history({ verbose: true }).map(move => ({
    color: move.color,
    from: move.from,
    to: move.to,
    san: move.san,
    piece: move.piece,
    captured: move.captured || null,
    promotion: move.promotion || null
  }));

  return {
    code: room.code,
    fen: game.fen(),
    pgn: game.pgn(),
    turn: game.turn(),
    history,

    players: {
      w: room.players.w
        ? {
            name: room.players.w.name,
            connected: room.players.w.connected
          }
        : null,

      b: room.players.b
        ? {
            name: room.players.b.name,
            connected: room.players.b.connected
          }
        : null
    },

    status:
      room.resigned || room.drawAgreed
        ? room.drawAgreed
          ? "draw"
          : "resigned"
        : game.isCheckmate()
          ? "checkmate"
          : game.isStalemate()
            ? "stalemate"
            : game.isThreefoldRepetition()
              ? "threefold"
              : game.isInsufficientMaterial()
                ? "insufficient"
                : game.isDraw()
                  ? "draw"
                  : game.isCheck()
                    ? "check"
                    : "playing",

    winner: room.resigned
      ? room.resigned === "w"
        ? "b"
        : "w"
      : game.isCheckmate()
        ? game.turn() === "w"
          ? "b"
          : "w"
        : null,

    drawOffer: room.drawOffer,
    rematch: room.rematch,
    timeLeft: room.timeLeft,

    running:
      !!(
        room.players.w?.connected &&
        room.players.b?.connected
      ) && !room.game.isGameOver()
  };
}

function emitState(room) {
  io.to(room.code).emit("state", publicState(room));
}

function getPlayer(room, socket) {
  if (!room) return null;

  for (const color of ["w", "b"]) {
    const player = room.players[color];

    if (player && player.socketId === socket.id) {
      return color;
    }
  }

  return null;
}

function joinSocket(socket, room, color, token) {
  const player = room.players[color];

  player.socketId = socket.id;
  player.connected = true;
  player.token = token || player.token || crypto.randomUUID();

  socket.join(room.code);
  socket.data.roomCode = room.code;
  socket.data.playerToken = player.token;

  socket.emit("joined", {
    code: room.code,
    color,
    token: player.token,
    state: publicState(room)
  });

  emitState(room);
}

function sendError(socket, message) {
  socket.emit("gameError", message);
}

async function saveGameRecord(room) {
  if (!db || room.saved) return;

  room.saved = true;

  try {
    const startedAt = room.createdAt
      ? new Date(room.createdAt).toISOString()
      : null;

    const history = room.game.history({ verbose: true })
      .map(move => move.san)
      .join(" ");

    const duration = room.createdAt
      ? Math.floor((Date.now() - room.createdAt) / 1000)
      : 0;

    let result = "unknown";
    let winner = null;

    if (room.drawAgreed) {
      result = "draw";
    } else if (room.resigned) {
      result = "resigned";
      winner = room.resigned === "w" ? "black" : "white";
    } else if (room.game.isCheckmate()) {
      result = "checkmate";
      winner = room.game.turn() === "w" ? "black" : "white";
    } else if (room.game.isDraw()) {
      result = "draw";
    }

    await db.execute({
      sql: `INSERT INTO chess_games
        (code, player_white, player_black, winner, result, moves,
         move_times, duration_seconds, time_white, time_black, started_at)
        VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?)`,
      args: [
        room.code,
        room.players.w?.name || "White",
        room.players.b?.name || "Black",
        winner,
        result,
        history,
        JSON.stringify(room.moveTimes || []),
        duration,
        room.timeLeft?.w ?? 0,
        room.timeLeft?.b ?? 0,
        startedAt
      ]
    });

    console.log("Game saved:", room.code, result);
  } catch (err) {
    console.error("Failed to save game:", err.message);
  }
}

io.on("connection", socket => {

  socket.on("createRoom", ({ name, token } = {}) => {
    const code = makeCode();

    const room = {
      code,
      game: new Chess(),
      drawOffer: null,
      rematch: { w: false, b: false },
      players: { w: null, b: null },
      createdAt: Date.now(),
      timeLeft: { w: 600, b: 600 },
      moveTimes: [],
      lastTick: Date.now()
    };

    room.players.w = {
      name: cleanName(name),
      token: token || crypto.randomUUID(),
      socketId: socket.id,
      connected: true
    };

    rooms.set(code, room);
    joinSocket(socket, room, "w", token);
  });

  socket.on("joinRoom", ({ code, name, token } = {}) => {
    code = String(code || "").trim().toUpperCase();

    const room = rooms.get(code);

    if (!room) {
      return sendError(
        socket,
        "Room not found. Check the code or create a new room."
      );
    }

    let existingColor = null;

    for (const color of ["w", "b"]) {
      if (
        room.players[color] &&
        token &&
        room.players[color].token === token
      ) {
        existingColor = color;
      }
    }

    if (existingColor) {
      return joinSocket(socket, room, existingColor, token);
    }

    if (!room.players.w) {
      return joinSocket(socket, room, "w", token);
    }

    if (!room.players.b) {
      room.players.b = {
        name: cleanName(name),
        token: token || crypto.randomUUID(),
        socketId: null,
        connected: false
      };

      return joinSocket(socket, room, "b", token);
    }

    return sendError(socket, "This room already has two players.");
  });

  socket.on("reconnectRoom", ({ code, token } = {}) => {
    code = String(code || "").trim().toUpperCase();

    const room = rooms.get(code);

    if (!room || !token) {
      return sendError(
        socket,
        "That game session is no longer available. Create a new room."
      );
    }

    for (const color of ["w", "b"]) {
      if (room.players[color] && room.players[color].token === token) {
        return joinSocket(socket, room, color, token);
      }
    }

    sendError(socket, "This player session was not found in that room.");
  });

  socket.on("spectate", ({ code } = {}) => {
    code = String(code || "").trim().toUpperCase();

    const room = rooms.get(code);

    if (!room) return sendError(socket, "Room not found.");

    socket.join(room.code);
    socket.data.roomCode = room.code;
    socket.data.isSpectator = true;

    socket.emit("spectating", {
      code,
      state: publicState(room)
    });
  });

  socket.on("move", ({ from, to, promotion } = {}) => {
    const room = rooms.get(socket.data.roomCode);
    const color = getPlayer(room, socket);

    if (!room || !color) {
      return sendError(socket, "Join a room before moving.");
    }

    if (!room.players.w || !room.players.b) {
      return sendError(socket, "Waiting for an opponent to join.");
    }

    if (room.game.isGameOver()) {
      return sendError(socket, "This game has ended.");
    }

    if (room.game.turn() !== color) {
      return sendError(socket, "It is not your turn.");
    }

    try {
      const move = room.game.move({
        from,
        to,
        promotion: promotion || "q"
      });

      room.moveTimes.push(Date.now() - room.createdAt);
      room.drawOffer = null;
      room.rematch = { w: false, b: false };

      io.to(room.code).emit("lastMove", {
        from: move.from,
        to: move.to,
        san: move.san

      }

      room.drawAgreed = true;

      `${color === "w" ? "White" : "Black"} resigned.`
    );

  socket.on("rematch", () => {

    const color = getPlayer(room, socket);

    if (!room || !color) return;

    room.rematch[color] = true;

    if (room.rematch.w && room.rematch.b) {

      room.rematch = { w: false, b: false };

      room.drawAgreed = false;

      room.lastTick = Date.now();
      room.createdAt = Date.now();

    } else {

      io.to(room.code).emit(
        "notice",
        "Rematch requested. Waiting for the other player."
      );
    }
  });

  // CHAT: Only one handler broadcasts each message.
  socket.on("chatMessage", ({ code, message } = {}) => {
    if (!code || typeof message !== "string") return;

    code = String(code).trim().toUpperCase();

    const cleanMsg = message.trim().slice(0, 256);

    if (!cleanMsg) return;

    const room = rooms.get(code);

    if (!room) return;

    let senderName = "Player";

    if (room.players.w && room.players.w.socketId === socket.id) {
      senderName = room.players.w.name;
    } else if (room.players.b && room.players.b.socketId === socket.id) {
      senderName = room.players.b.name;
    } else if (socket.data.roomCode === code) {
      senderName = "Spectator/Player";
    } else {
      return;
    }

    io.to(code).emit("chatMessage", {
      name: senderName,
      message: cleanMsg,
      time: Date.now()
    });
  });

  socket.on("disconnect", () => {
    const room = rooms.get(socket.data.roomCode);
    const color = getPlayer(room, socket);

    if (room && color && room.players[color]) {
      room.players[color].connected = false;
      room.players[color].socketId = null;

      emitState(room);
    }
  });
});

// Chess clock.
setInterval(() => {
  const now = Date.now();

  for (const room of rooms.values()) {
    if (
      !room.players.w ||
      !room.players.b ||
      !room.players.w.connected ||
      !room.players.b.connected
    ) {
      room.lastTick = now;
      continue;
    }

    if (room.game.isGameOver()) {
      room.lastTick = now;
      continue;
    }

    const dt = Math.max(
      0,
      Math.round((now - room.lastTick) / 1000)
    );

    if (dt <= 0) continue;

    room.lastTick = now;

    const turn = room.game.turn();

    room.timeLeft[turn] = Math.max(
      0,
      room.timeLeft[turn] - dt
    );

    if (room.timeLeft[turn] === 0) {
      room.resigned = turn;

      room.game.header(
        "Result",
        turn === "w" ? "0-1" : "1-0"
      );

      emitState(room);

      io.to(room.code).emit(
        "notice",
        (turn === "w" ? "White" : "Black") + " ran out of time."
      );

      saveGameRecord(room);
    } else {
      io.to(room.code).emit("clock", {
        timeLeft: room.timeLeft,
        turn
      });
    }
  }
}, 1000);
